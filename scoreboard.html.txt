                </div>

    <script type="module">
        import { initializeApp } from 'firebase/app';
        import { getAuth, signInAnonymously, onAuthStateChanged } from 'firebase/auth';
        import { getFirestore, collection, doc, onSnapshot, setDoc, updateDoc, deleteDoc, query } from 'firebase/firestore';

        // --- 這裡請填入您的 Firebase 配置 ---
        const firebaseConfig = {
            apiKey: "", // 保持為空，環境會自動填入或請填入您自己的
            authDomain: "",
            projectId: "",
            storageBucket: "",
            messagingSenderId: "",
            appId: ""
        };

        // 如果在特殊環境下，嘗試抓取環境變數
        const actualConfig = typeof __firebase_config !== 'undefined' ? JSON.parse(__firebase_config) : firebaseConfig;
        const appIdStr = typeof __app_id !== 'undefined' ? __app_id : 'my-scoreboard-app';

        const app = initializeApp(actualConfig);
        const auth = getAuth(app);
        const db = getFirestore(app);

        let user = null;
        let players = [];

        // 初始化
        async function init() {
            await signInAnonymously(auth);
            onAuthStateChanged(auth, (currUser) => {
                user = currUser;
                if (user) startListening();
            });
            lucide.createIcons();
        }

        // 監聽雲端數據
        function startListening() {
            const path = `artifacts/${appIdStr}/users/${user.uid}/players`;
            const q = collection(db, path);
            
            onSnapshot(q, (snapshot) => {
                players = snapshot.docs.map(d => ({ id: d.id, ...d.data() }))
                    .sort((a, b) => (a.createdAt || 0) - (b.createdAt || 0));
                
                render();
                document.getElementById('loader').classList.add('hidden');
                document.getElementById('main-content').classList.remove('hidden');
            });
        }

        // 渲染畫面
        window.render = () => {
            const list = document.getElementById('playerList');
            list.innerHTML = players.length === 0 ? '<p class="text-center py-10 text-slate-300">目前沒有計分項目</p>' : '';
            
            players.forEach(p => {
                const div = document.createElement('div');
                div.className = 'bg-white p-6 rounded-[2.5rem] shadow-md border border-slate-100 flex items-center justify-between score-card';
                div.innerHTML = `
                    <div class="flex-1">
                        <input type="text" value="${p.name}" onchange="updateName('${p.id}', this.value)" class="text-[10px] font-black text-slate-400 bg-transparent outline-none uppercase tracking-widest mb-1">
                        <div class="text-5xl font-black text-slate-800 tabular-nums">${p.score}</div>
                    </div>
                    <div class="flex items-center gap-2">
                        <button onclick="changeScore('${p.id}', -1)" class="w-14 h-14 rounded-3xl bg-slate-50 flex items-center justify-center text-rose-500 active:bg-rose-100 transition-colors">
                            <i data-lucide="minus" class="w-6 h-6"></i>
                        </button>
                        <button onclick="changeScore('${p.id}', 1)" class="w-14 h-14 rounded-3xl bg-indigo-600 flex items-center justify-center text-white active:bg-indigo-700 shadow-lg shadow-indigo-100">
                            <i data-lucide="plus" class="w-6 h-6"></i>
                        </button>
                    </div>
                `;
                list.appendChild(div);
            });
            lucide.createIcons();
        };

        // 操作函式
        window.changeScore = async (id, dir) => {
            const step = parseInt(document.getElementById('stepValue').value) || 1;
            const p = players.find(x => x.id === id);
            const ref = doc(db, `artifacts/${appIdStr}/users/${user.uid}/players`, id);
            await updateDoc(ref, { score: p.score + (dir * step) });
        };

        window.updateName = async (id, name) => {
            const ref = doc(db, `artifacts/${appIdStr}/users/${user.uid}/players`, id);
            await updateDoc(ref, { name });
        };

        window.addPlayer = async () => {
            const id = Date.now().toString();
            const ref = doc(db, `artifacts/${appIdStr}/users/${user.uid}/players`, id);
            await setDoc(ref, { name: '新對象', score: 0, createdAt: Date.now() });
        };

        window.resetAll = async () => {
            if (!confirm('確定要全部歸零？')) return;
            for (const p of players) {
                const ref = doc(db, `artifacts/${appIdStr}/users/${user.uid}/players`, p.id);
                await updateDoc(ref, { score: 0 });
            }
        };

        // 註冊 Service Worker (PWA)
        if ('serviceWorker' in navigator) {
            const sw = `self.addEventListener('fetch', e => e.respondWith(fetch(e.request)));`;
            const blob = new Blob([sw], { type: 'text/javascript' });
            navigator.serviceWorker.register(URL.createObjectURL(blob));
        }

        init();
    </script>
